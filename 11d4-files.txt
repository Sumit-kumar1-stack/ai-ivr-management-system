//--------------------------------------------------
// Configuration
//--------------------------------------------------

const DEFAULT_RERANK_CANDIDATE_LIMIT =
  8;

const configuredRerankCandidateLimit =
  Number(
    process.env
      .KNOWLEDGE_RERANK_CANDIDATE_LIMIT
  );

const RERANK_CANDIDATE_LIMIT =
  Number.isInteger(
    configuredRerankCandidateLimit
  ) &&
  configuredRerankCandidateLimit >=
    3 &&
  configuredRerankCandidateLimit <=
    20
    ? configuredRerankCandidateLimit
    : DEFAULT_RERANK_CANDIDATE_LIMIT;

/*
 * Conservative default:
 *
 * Keep AI reranking enabled until we have enough
 * production/test evidence that deterministic BM25
 * dominance is safe for the bank's knowledge base.
 */
const ENABLE_CONFIDENCE_SKIP =
  process.env
    .KNOWLEDGE_CONFIDENCE_SKIP_RERANK ===
  "true";

const DEFAULT_DOMINANCE_RATIO =
  2.25;

const configuredDominanceRatio =
  Number(
    process.env
      .KNOWLEDGE_BM25_DOMINANCE_RATIO
  );

const BM25_DOMINANCE_RATIO =
  Number.isFinite(
    configuredDominanceRatio
  ) &&
  configuredDominanceRatio >=
    1.2 &&
  configuredDominanceRatio <=
    10
    ? configuredDominanceRatio
    : DEFAULT_DOMINANCE_RATIO;

//--------------------------------------------------
// Confidence Gate
//--------------------------------------------------

function canSkipReranking(
  candidates: Array<{
    score: number;
  }>
): boolean {
  if (
    !ENABLE_CONFIDENCE_SKIP
  ) {
    return false;
  }

  if (
    candidates.length <
    2
  ) {
    return true;
  }

  const first =
    candidates[0];

  const second =
    candidates[1];

  if (
    !first ||
    !second
  ) {
    return false;
  }

  if (
    !Number.isFinite(
      first.score
    ) ||
    !Number.isFinite(
      second.score
    ) ||
    first.score <=
      0 ||
    second.score <=
      0
  ) {
    return false;
  }

  const dominanceRatio =
    first.score /
    second.score;

  return (
    dominanceRatio >=
    BM25_DOMINANCE_RATIO
  );
}

//--------------------------------------------------
// Retrieve Knowledge
//--------------------------------------------------

export async function retrieveKnowledge(
  question: string,
  limit = 5
): Promise<
  RetrievedKnowledgeChunk[]
> {
  const startedAt =
    process.hrtime.bigint();

  const normalizedQuestion =
    question.trim();

  //--------------------------------------------------
  // Empty Query
  //--------------------------------------------------

  if (
    !normalizedQuestion
  ) {
    log.debug(
      {
        event:
          "knowledge.retrieval.skipped",

        reason:
          "empty_question",
      },
      "Knowledge retrieval skipped"
    );

    return [];
  }

  //--------------------------------------------------
  // Normalize Requested Result Count
  //--------------------------------------------------

  const normalizedLimit =
    Math.max(
      1,
      Math.min(
        limit,
        20
      )
    );

  try {
    //--------------------------------------------------
    // Load Knowledge Chunks
    //--------------------------------------------------

    const chunks =
      await prisma
        .knowledgeChunk
        .findMany();

    log.debug(
      {
        event:
          "knowledge.retrieval.started",

        queryCharacterCount:
          normalizedQuestion.length,

        availableChunkCount:
          chunks.length,

        requestedLimit:
          normalizedLimit,

        rerankCandidateLimit:
          RERANK_CANDIDATE_LIMIT,

        confidenceSkipEnabled:
          ENABLE_CONFIDENCE_SKIP,
      },
      "Knowledge retrieval started"
    );

    //--------------------------------------------------
    // BM25
    //--------------------------------------------------

    const bm25StartedAt =
      process.hrtime.bigint();

    const scored =
      chunks
        .map(
          chunk => ({
            ...chunk,

            score:
              bm25Score(
                normalizedQuestion,
                chunk.content
              ),
          })
        )
        .filter(
          chunk =>
            Number.isFinite(
              chunk.score
            ) &&
            chunk.score >
              0
        )
        .sort(
          (
            first,
            second
          ) =>
            second.score -
            first.score
        );

    const bm25Ms =
      getDurationMs(
        bm25StartedAt
      );

    log.debug(
      {
        event:
          "knowledge.retrieval.bm25_completed",

        availableChunkCount:
          chunks.length,

        candidateCount:
          scored.length,

        highestScore:
          scored[0]?.score ??
          null,

        secondHighestScore:
          scored[1]?.score ??
          null,

        bm25Ms,
      },
      "BM25 knowledge search completed"
    );

    //--------------------------------------------------
    // No Candidates
    //--------------------------------------------------

    if (
      scored.length ===
      0
    ) {
      log.info(
        {
          event:
            "knowledge.retrieval.no_candidates",

          queryCharacterCount:
            normalizedQuestion.length,

          bm25Ms,

          durationMs:
            getDurationMs(
              startedAt
            ),
        },
        "No relevant BM25 candidates found"
      );

      return [];
    }

    //--------------------------------------------------
    // Single Candidate
    //--------------------------------------------------

    if (
      scored.length ===
      1
    ) {
      const onlyCandidate =
        scored[0];

      log.info(
        {
          event:
            "knowledge.retrieval.rerank_skipped",

          reason:
            "single_candidate",

          candidateCount:
            1,

          bm25Ms,

          durationMs:
            getDurationMs(
              startedAt
            ),
        },
        "Knowledge reranking skipped"
      );

      return [
        {
          content:
            onlyCandidate.content,

          score:
            onlyCandidate.score,

          documentId:
            onlyCandidate.documentId,

          chunkIndex:
            onlyCandidate.chunkIndex,
        },
      ];
    }

    //--------------------------------------------------
    // Bound AI Reranking Candidates
    //--------------------------------------------------

    /*
     * Previously up to 20 BM25 matches could be sent
     * to the AI reranker.
     *
     * Keep only the strongest candidates before the
     * expensive reranking stage.
     */
    const rerankCandidates =
      scored.slice(
        0,
        RERANK_CANDIDATE_LIMIT
      );

    //--------------------------------------------------
    // Optional High-Confidence BM25 Fast Path
    //--------------------------------------------------

    if (
      canSkipReranking(
        rerankCandidates
      )
    ) {
      const topResults =
        rerankCandidates
          .slice(
            0,
            normalizedLimit
          )
          .map(
            chunk => ({
              content:
                chunk.content,

              score:
                chunk.score,

              documentId:
                chunk.documentId,

              chunkIndex:
                chunk.chunkIndex,
            })
          );

      const first =
        rerankCandidates[0];

      const second =
        rerankCandidates[1];

      const dominanceRatio =
        first &&
        second &&
        second.score >
          0
          ? first.score /
            second.score
          : null;

      log.info(
        {
          event:
            "knowledge.retrieval.rerank_skipped",

          reason:
            "high_bm25_confidence",

          originalCandidateCount:
            scored.length,

          boundedCandidateCount:
            rerankCandidates.length,

          returnedChunkCount:
            topResults.length,

          highestScore:
            first?.score ??
            null,

          secondHighestScore:
            second?.score ??
            null,

          dominanceRatio,

          configuredDominanceRatio:
            BM25_DOMINANCE_RATIO,

          bm25Ms,

          durationMs:
            getDurationMs(
              startedAt
            ),
        },
        "High-confidence BM25 result used without AI reranking"
      );

      return topResults;
    }

    //--------------------------------------------------
    // AI Re-ranking
    //--------------------------------------------------

    const rerankStartedAt =
      process.hrtime.bigint();

    const reranked =
      await rerankKnowledge(
        normalizedQuestion,
        rerankCandidates
      );

    const rerankMs =
      getDurationMs(
        rerankStartedAt
      );

    //--------------------------------------------------
    // Final Results
    //--------------------------------------------------

    const results =
      reranked
        .slice(
          0,
          normalizedLimit
        )
        .map(
          chunk => ({
            content:
              chunk.content,

            score:
              chunk.score,

            documentId:
              chunk.documentId,

            chunkIndex:
              chunk.chunkIndex,
          })
        );

    //--------------------------------------------------
    // Metrics
    //--------------------------------------------------

    log.info(
      {
        event:
          "knowledge.retrieval.completed",

        bm25CandidateCount:
          scored.length,

        rerankCandidateCount:
          rerankCandidates.length,

        rerankedCandidateCount:
          reranked.length,

        returnedChunkCount:
          results.length,

        highestBm25Score:
          scored[0]?.score ??
          null,

        bm25Ms,

        rerankMs,

        durationMs:
          getDurationMs(
            startedAt
          ),
      },
      "Knowledge retrieval completed"
    );

    return results;
  } catch (
    error
  ) {
    log.error(
      {
        event:
          "knowledge.retrieval.failed",

        queryCharacterCount:
          normalizedQuestion.length,

        durationMs:
          getDurationMs(
            startedAt
          ),

        error:
          normalizeError(
            error
          ),
      },
      "Knowledge retrieval failed"
    );

    throw error;
  }
}import {

//--------------------------------------------------
// Index Knowledge Chunks
//--------------------------------------------------

export async function indexDocuments(
  documentId: string,
  chunks: string[]
): Promise<void> {
  const startedAt =
    process.hrtime.bigint();

  if (
    chunks.length ===
    0
  ) {
    log.debug(
      {
        event:
          "knowledge.indexing.skipped",

        documentId,

        reason:
          "no_chunks",
      },
      "Knowledge indexing skipped"
    );

    return;
  }

  log.info(
    {
      event:
        "knowledge.indexing.started",

      documentId,

      chunkCount:
        chunks.length,

      totalCharacterCount:
        chunks.reduce(
          (
            total,
            chunk
          ) =>
            total +
            chunk.length,
          0
        ),
    },
    "Knowledge chunk indexing started"
  );

  try {
    await prisma.$transaction(
      chunks.map(
        (
          chunk,
          index
        ) =>
          prisma.knowledgeChunk.create({
            data: {
              documentId,

              chunkIndex:
                index,

              content:
                chunk,
            },
          })
      )
    );

    log.info(
      {
        event:
          "knowledge.indexing.completed",

        documentId,

        chunkCount:
          chunks.length,

        durationMs:
          getDurationMs(
            startedAt
          ),
      },
      "Knowledge chunk indexing completed"
    );
  } catch (
    error
  ) {
    log.error(
      {
        event:
          "knowledge.indexing.failed",

        documentId,

        chunkCount:
          chunks.length,

        durationMs:
          getDurationMs(
            startedAt
          ),

        error:
          normalizeError(
            error
          ),
      },
      "Knowledge chunk indexing failed"
    );

    throw error;
  }
}import { prisma } from "@/lib/prisma";
